==================== 附：可复现的比对命令（在任意能联网的环境）====================

  python3 - <<'PY'
  import urllib.request, json, hashlib, os
  tree=json.load(urllib.request.urlopen(
      "https://api.github.com/repos/yanyuan-cn/xiaoman/git/trees/main?recursive=1"))
  remote={n['path']:n['sha'] for n in tree['tree'] if n['type']=='blob'}
  def sha(p):
      b=open(p,'rb').read()
      return hashlib.sha1(b"blob %d\0"%len(b)+b).hexdigest()
  local={}
  for r,d,fs in os.walk('.'):
      d[:]=[x for x in d if x not in ('.git','build','.dart_tool','android')]
      for f in fs:
          p=os.path.relpath(os.path.join(r,f),'.').replace(os.sep,'/')
          local[p]=sha(os.path.join(r,f))
  print("same", sum(1 for k in local if remote.get(k)==local[k]))
  print("diff", [k for k in local if remote.get(k) not in (None, local[k])])
  print("missing_remote", [k for k in local if k not in remote])
  PY

不需要 git 命令，纯 HTTP + 本地哈希。
